\section{Kiến trúc Pipeline AI 2 lớp (Dual-Layer AI Resilience)}
Nhằm khắc phục nhược điểm phụ thuộc đường truyền và chi phí API điện toán đám mây, hệ thống triển khai kiến trúc xử lý kép (Hình~\ref{fig:ai_pipeline}):
\begin{itemize}
    \item \textbf{Lớp 1 (Cloud LLM):} Gọi Google Gemini API với System Prompt khắt khe và Ground-Truth Context Injection.
    \item \textbf{Lớp 2 (Local Deterministic Rule Engine):} Tự động kích hoạt khi mất kết nối mạng hoặc thiếu API Key, tính toán trực tiếp số ngày cạn tồn và phát hiện bất thường dựa trên quy tắc toán học xác định.
\end{itemize}

\section{Mô hình toán học Burn-rate và Phát hiện Bất thường}
\begin{enumerate}
    \item \textbf{Tốc độ xuất bình quân ngày (Daily Burn Rate - $V$):}
    \begin{equation}
        V = \frac{\sum_{i=1}^{30} \text{QuantityExported}_i}{30}
    \end{equation}
    \item \textbf{Số ngày cạn tồn dự kiến (Depletion Days - $D$):}
    \begin{equation}
        D = \begin{cases}
            \frac{\text{CurrentStock}}{V}, & \text{khi } V > 0 \\
            \infty \text{ (Tồn an toàn)}, & \text{khi } V = 0
        \end{cases}
    \end{equation}
    Quy tắc này loại bỏ hoàn toàn lỗi ZeroDivisionError.
    \item \textbf{Kiểm định dị biệt Z-score:}
    \begin{equation}
        Z = \frac{X - \mu}{\sigma}
    \end{equation}
    Khi $|Z| > 2.5$, hệ thống gắn cờ cảnh báo giao dịch bất thường (Hình~\ref{fig:anomaly}).
\end{enumerate}
